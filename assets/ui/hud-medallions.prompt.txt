#!/usr/bin/env bash
set -euo pipefail
hud_work_dir=$(mktemp -d /tmp/casino-classic-hud.XXXXXX)
trap 'rm -rf "$hud_work_dir"' EXIT
magick -size 1536x1024 xc:none "PNG32:$hud_work_dir/atlas.png"
magick -size 48x48 xc:none -fill '#090b0ee6' -stroke '#777973' -strokewidth 1 -draw 'circle 24,24 24,2' -fill none -stroke '#25282b' -draw 'circle 24,24 24,4' "PNG32:$hud_work_dir/frame.png"
for hud_y in 24 508; do
    for hud_x in 56 536 1016; do
        magick assets/ui/hud-items-source.png -crop "464x464+$hud_x+$hud_y" +repage -filter Box -resize '40x40!' "PNG32:$hud_work_dir/object.png"
        magick "$hud_work_dir/object.png" -alpha extract -threshold 50% "$hud_work_dir/mask.png"
        magick "$hud_work_dir/object.png" -background '#090b0e' -alpha remove -alpha off -dither FloydSteinberg -colors 32 "$hud_work_dir/mask.png" -alpha off -compose CopyOpacity -composite "PNG32:$hud_work_dir/sprite.png"
        magick "$hud_work_dir/frame.png" "$hud_work_dir/sprite.png" -gravity center -compose Over -composite -filter point -resize '464x464!' "PNG32:$hud_work_dir/badge.png"
        magick "$hud_work_dir/atlas.png" "$hud_work_dir/badge.png" -geometry "+$hud_x+$hud_y" -compose Over -composite "PNG32:$hud_work_dir/next.png"
        mv "$hud_work_dir/next.png" "$hud_work_dir/atlas.png"
    done
done
magick "$hud_work_dir/atlas.png" PNG32:assets/ui/hud-medallions.png
python3 - <<'PY'
import struct
from pathlib import Path
png=Path('assets/ui/hud-medallions.png').read_bytes()
assert png[:8]==b'\x89PNG\r\n\x1a\n'
assert struct.unpack('>II',png[16:24])==(1536,1024)
assert png[25]==6
print('Classic HUD: RGBA atlas and original crop dimensions verified')
PY
